## All proprietary files from this list are from the OnePlus 7 Pro
## (GM1911_11.H.41, ColorOS 12.1 / Android 12).
##
## Stock layout note: the camera and camera service live on /my_product and
## the gallery on /my_bigball. This tree has neither partition, so everything
## is installed under system_ext.

# Camera
my_product/priv-app/OnePlusCamera/OnePlusCamera.apk:system_ext/priv-app/OnePlusCamera/OnePlusCamera.apk
my_product/priv-app/OnePlusCameraService/OnePlusCameraService.apk:system_ext/priv-app/OnePlusCameraService/OnePlusCameraService.apk
my_bigball/priv-app/OnePlusGallery/OnePlusGallery.apk:system_ext/priv-app/OnePlusGallery/OnePlusGallery.apk

# Camera extension loaded by cameraserver. Kept in system_ext/lib64 so the
# dlopen path in the CameraService patch (system_ext/lib64/libcsextimpl.so)
# resolves without a /vendor or /odm lookup.
system_ext/lib64/libcsextimpl.so:system_ext/lib64/libcsextimpl.so

# HIDL interfaces the extension links against.
system_ext/lib64/vendor.oplus.hardware.cameraMDM@1.0.so:system_ext/lib64/vendor.oplus.hardware.cameraMDM@1.0.so
system_ext/lib64/vendor.oplus.hardware.cameraMDM@2.0.so:system_ext/lib64/vendor.oplus.hardware.cameraMDM@2.0.so
system_ext/lib64/vendor.oplus.hardware.commondcs@1.0.so:system_ext/lib64/vendor.oplus.hardware.commondcs@1.0.so
# HIDL performance interface. libcsextimpl.so has a DT_NEEDED on it, and this
# tree only ships the AIDL replacement, whose SONAME differs.
system/lib64/vendor.oplus.hardware.performance@1.0.so:system_ext/lib64/vendor.oplus.hardware.performance@1.0.so

# Nothing else from /my_product/lib64 is shipped.
#
# The camera and gallery bundle every library they dlopen: libOnePlusCamera and
# libCameraNext implement the APS JNI, and the ARCSoft, Morpho and blur pieces
# are APK assets. A dex scan of both APKs finds no loadLibrary() call for any
# /my_product name, and libcsextimpl's DT_NEEDED list does not mention them
# either, so installing them would only add unused files to system_ext.
